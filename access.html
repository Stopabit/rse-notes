2  Accessing & Manipulating an RSE

2.1 Creating an RSE from scratch

Although in practice you will load an RSE produced by a pipeline (see Chapter 3), it is instructive to build one manually.

library(SummarizedExperiment)
library(GenomicRanges)

## ── 1. Assay matrices ───────────────────────────────────────────────────────
counts <- matrix(
  rpois(200, lambda = 500),   # 20 genes × 10 samples
  nrow = 20, ncol = 10,
  dimnames = list(
    paste0("ENSG", 1:20),     # row names = gene IDs
    paste0("S", 1:10)         # col names = sample IDs
  )
)

normcounts <- log2(counts + 1)  # a second assay layer

## ── 2. rowRanges (GRanges) ──────────────────────────────────────────────────
gr <- GRanges(
  seqnames = c(rep("chr1", 10), rep("chr2", 10)),
  ranges   = IRanges(start = seq(1e4, 2e5, length.out = 20),
                     width = 1000),
  strand   = rep(c("+", "-"), 10),
  gene_id  = paste0("ENSG", 1:20),   # mcols
  symbol   = paste0("GENE", 1:20)    # mcols
)
names(gr) <- paste0("ENSG", 1:20)   # must match rownames(counts)

## ── 3. colData ───────────────────────────────────────────────────────────────
col_df <- DataFrame(
  sample    = paste0("S", 1:10),
  condition = rep(c("treated", "control"), each = 5),
  batch     = rep(c("B1", "B2"), 5),
  sex       = sample(c("M", "F"), 10, replace = TRUE)
)
rownames(col_df) <- paste0("S", 1:10)   # must match colnames(counts)

## ── 4. Assemble ──────────────────────────────────────────────────────────────
rse <- SummarizedExperiment(
  assays    = list(counts = counts, normcounts = normcounts),
  rowRanges = gr,
  colData   = col_df,
  metadata  = list(organism = "Homo sapiens", pipeline = "tutorial")
)

rse   # inspect
TipKey: matching names

rownames(counts) must equal names(gr);
colnames(counts) must equal rownames(col_df).
If they don’t match, the constructor throws an error.

2.2 Reading slots

2.2.1 Assays

# All assay names
assayNames(rse)
# [1] "counts"     "normcounts"

# Get the raw count matrix (returns a plain matrix)
counts_mat <- assay(rse, "counts")
counts_mat <- assay(rse, 1)         # same — by position

# Get all assays as a SimpleList
all_assays <- assays(rse)

# Add a new assay
assay(rse, "logCPM") <- edgeR::cpm(counts_mat, log = TRUE)

2.2.2 rowRanges / rowData

# Full GRanges
rowRanges(rse)            # returns GRanges

# Metadata columns of the GRanges (= rowData)
rowData(rse)              # returns DataFrame
rowData(rse)$symbol       # single column as a vector

# Access via GRanges methods
seqnames(rowRanges(rse))  # chromosome of every feature
start(rowRanges(rse))     # start positions
width(rowRanges(rse))     # feature widths in bp
strand(rowRanges(rse))    # strand

2.2.3 colData

colData(rse)              # full DataFrame
colData(rse)$condition    # one column

# Shorthand with $ (very convenient!)
rse$condition             # identical to colData(rse)$condition
rse$batch

# Add a new sample annotation
rse$RIN <- c(9.1, 8.7, 9.3, 8.9, 9.0,
             8.5, 9.2, 8.8, 9.1, 8.6)  # RNA integrity

2.2.4 metadata

metadata(rse)             # returns a named list
metadata(rse)$organism    # one element
metadata(rse)$QC <- list(minRIN = 8, minDepth = 1e6)

2.3 Subsetting

Subsetting uses standard [row, col] syntax — just like a matrix.

## By index
rse[1:5, 1:3]

## By name
rse["ENSG3", "S2"]

## By logical vector — keep treated samples only
rse[, rse$condition == "treated"]

## Keep only chr1 features
rse[seqnames(rowRanges(rse)) == "chr1", ]

## Combine both
rse_sub <- rse[
  seqnames(rowRanges(rse)) == "chr1",
  rse$condition == "treated"
]
dim(rse_sub)

2.4 Merging RSEs — cbind and rbind

# Combine samples (same features, different samples)
rse_combined <- cbind(rse_batch1, rse_batch2)

# Combine features (same samples, different genomic regions)
rse_all <- rbind(rse_chr1, rse_chr2)
Warning

For cbind, both objects must have identical rowRanges (same features).
For rbind, both objects must have identical colData (same samples).

2.5 Useful utility functions

# How many assays?
length(assays(rse))

# Feature names and dimensions
rownames(rse); colnames(rse); dim(rse)

# Quick summary of colData
summary(as.data.frame(colData(rse)))

# Which chromosomes are represented?
table(seqnames(rowRanges(rse)))

# Find features overlapping a region of interest
library(GenomicRanges)
query  <- GRanges("chr1", IRanges(1e4, 5e4))
hits   <- findOverlaps(rowRanges(rse), query)
rse_roi <- rse[queryHits(hits), ]
